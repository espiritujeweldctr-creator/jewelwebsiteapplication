My Java Project

Word Guessing Game

Programming Language: Java

The Word Guessing Game is a simple console-based game developed using Java. The player needs to guess the hidden word by reading the given hint. Each word has five attempts, and the player earns points for every correct answer.

Game Features

  • Five hidden words to guess.
  • Hints for every hidden word.
  • Five attempts for each word.
  • Points and scoring system.
  • Option to play again.

Code

import java.util.Scanner;
public class Main {
public static void main(String[] args) {
Scanner input = new Scanner(System.in); String[] words = {"apple", "banana", "computer", "school", "guitar"};
String[] hints = {
"A red or green fruit",
"A yellow fruit",
"An electronic device",
"A place where students learn",
"A musical instrument"
};
int score = 0;
int attempts;
String guess;
String playAgain = "yes";
System.out.println("================================");
System.out.println(" WORD GUESSING GAME");
System.out.println("================================");
System.out.println("Instructions:");
System.out.println("1. Guess the hidden word.");
System.out.println("2. Read the hint carefully.");
System.out.println("3. You have 5 attempts.");
System.out.println("4. Get points for every correct answer.");
while (playAgain.equalsIgnoreCase("yes")) {
for (int i = 0; i < words.length; i++) {
String word = words[i];
String hint = hints[i];
attempts = 5;
System.out.println("\nHint: " + hint);
System.out.println("The word has " + word.length() + " 
letters.");
while (attempts > 0) {
System.out.print("Enter your guess: ");
guess = input.nextLine();
if (guess.equalsIgnoreCase(word)) {
System.out.println("Correct! You got it!"); score += attempts;
break;
} else {
attempts--;
System.out.println("Wrong answer! Try again.");
if (attempts > 0) {
System.out.println("Attempts left: " + attempts);
} else {
System.out.println("Game Over!");
System.out.println("The correct word is: " + word);
}
}
}
}
System.out.println("\nYour current score: " + score);
System.out.print("Do you want to play again? (yes/no): ");
playAgain = input.nextLine();
if (playAgain.equalsIgnoreCase("yes")) {
score = 0;
}
}
System.out.println("\n================================");
System.out.println(" GAME OVER");
System.out.println("================================");
System.out.println("Your Final Score: " + score);
System.out.println("Thank you for playing!");
input.close();
}
}
